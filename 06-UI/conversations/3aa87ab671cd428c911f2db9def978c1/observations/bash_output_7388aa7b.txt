import { rawEventsQueue } from './src/queue';
async function test() {
  console.log('[pipeline] Testing Kulturhuset → Queue pipeline...\n');

  // Clear any existing Kulturhuset jobs from previous test
  console.log('[pipeline] Clearing existing jobs...');
  const countsBefore = await rawEventsQueue.getJobCounts();
  console.log('[pipeline] Queue counts before:', countsBefore);

  // Run Kulturhuset source
  console.log('\n[pipeline] Running Kulturhuset source...');
  const result = await runKulturhusetSource();
  console.log('\n[pipeline] Source result:', JSON.stringify(result, null, 2));

  // Check queue
  const countsAfter = await rawEventsQueue.getJobCounts();
  console.log('\n[pipeline] Queue counts after:', countsAfter);

  if (countsAfter.waiting > 0) {
    console.log('\n[pipeline] ✅ SUCCESS: Jobs are queued!');
  } else {
    console.log('\n[pipeline] ❌ FAIL: No jobs in queue');
  }
}
test().catch(err => {
  console.error('[pipeline] Error:', err.message);
  process.exit(1);
});

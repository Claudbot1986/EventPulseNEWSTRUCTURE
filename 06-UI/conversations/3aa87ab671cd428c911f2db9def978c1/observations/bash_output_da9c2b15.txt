import * as dotenv from 'dotenv';
dotenv.config({ override: true });
import { rawEventsQueue } from './src/queue';
import { createNormalizerWorker } from './src/queue';
import { processRawEvent } from './src/normalizer';
async function test() {
  console.log('[worker] Testing normalizer worker...\n');

  // Check queue before
  const countsBefore = await rawEventsQueue.getJobCounts();
  console.log('[worker] Queue counts before:', countsBefore);

  // Create worker
  console.log('\n[worker] Starting worker...');
  const worker = createNormalizerWorker(processRawEvent);

  worker.on('completed', (job) => {
    console.log(`[worker] ✅ Completed job: ${job.id}`);
  });

  worker.on('failed', (job, err) => {
    console.error(`[worker] ❌ Failed job: ${job?.id}`, err.message);
  });

  // Wait for jobs to process
  console.log('\n[worker] Waiting for jobs to process (10 seconds)...');
  await new Promise(resolve => setTimeout(resolve, 10000));

  // Check queue after
  const countsAfter = await rawEventsQueue.getJobCounts();
  console.log('\n[worker] Queue counts after:', countsAfter);

  // Cleanup
  await worker.close();
  console.log('\n[worker] Worker closed');

  if (countsAfter.completed > 0) {
    console.log('\n[worker] ✅ SUCCESS: Jobs processed!');
  } else {
    console.log('\n[worker] ❌ FAIL: No jobs completed');
  }
}
test().catch(err => {
  console.error('[worker] Error:', err.message);
  process.exit(1);
});
EOF

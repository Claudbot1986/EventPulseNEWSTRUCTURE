import { runKulturhusetSource } from './src/sources/sourceRunner';
async function main() {
  console.log('=== EventPulse Ingestion Test ===');

  // 1. Start normalizer worker
  const worker = createNormalizerWorker(processRawEvent);
  worker.on('completed', (job) => {
    // Silent completion
  });

  console.log('[✓] Normalizer worker started');
  // 2. Clear old Kulturhuset jobs from queue to avoid duplicates

  // Note: We don't clear ticketmaster/malmo-live as they're working fine

  const result = await runKulturhusetSource();
  console.log(`Kulturhuset: found=${result.eventsFound}, queued=${result.eventsQueued}, errors=${result.errors.length}`);
  // 4. Wait for queue to process
  console.log('\n[...] Waiting for queue to process (30 seconds)...');
  await new Promise(resolve => setTimeout(resolve, 30000));
  // 5. Check queue stats
  const counts = await rawEventsQueue.getJobCounts();
  console.log('\n=== Queue Stats ===');
  console.log(`Active: ${counts.active}, Completed: ${counts.completed}, Failed: ${counts.failed}, Waiting: ${counts.waiting}`);

  // Clean up
  await worker.close();
  console.log('\n[✓] Done');
}
main().catch(err => {

});
